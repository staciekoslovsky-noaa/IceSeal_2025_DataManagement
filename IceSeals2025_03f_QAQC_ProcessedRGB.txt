--RGB processed not in IR processed
SELECT flight, camera_view, COUNT(id) AS num_missing
FROM surv_ice_seals_2025.tbl_detections_processed_rgb
WHERE processed_detection_id NOT IN (SELECT processed_detection_id FROM surv_ice_seals_2025.tbl_detections_processed_ir)
GROUP BY flight, camera_view
ORDER BY flight, camera_view;

--IR processed not in RGB
SELECT flight, camera_view, COUNT(id) AS num_missing
FROM surv_ice_seals_2025.tbl_detections_processed_ir
WHERE processed_detection_id NOT IN (SELECT processed_detection_id FROM surv_ice_seals_2025.tbl_detections_processed_rgb)
AND (flight, camera_view) IN (SELECT flight, camera_view FROM annotations.summ_who_did_what 
							WHERE project_schema = 'surv_ice_seals_2025' AND processing_step = 'Species ID (RGB)' AND processing_completed_by IS NOT NULL)
GROUP BY flight, camera_view
ORDER BY flight, camera_view;

--Check for missing data
SELECT *
FROM surv_ice_seals_2025.tbl_detections_processed_rgb
WHERE detection_type IS NULL
OR species_confidence IS NULL
OR age_class IS NULL
OR age_class_confidence IS NULL
ORDER BY flight, camera_view;

SELECT flight, camera_view, detection as detection_id, image_name, detection_type, species_confidence, age_class, age_class_confidence, processing_completed_by
FROM surv_ice_seals_2025.tbl_detections_processed_rgb
INNER JOIN annotations.summ_who_did_what USING (flight, camera_view)
WHERE (detection_type IS NULL OR detection_type = 'NA'
OR species_confidence IS NULL OR species_confidence = 'NA'
OR age_class IS NULL OR age_class = 'NA'
OR age_class_confidence IS NULL OR age_class_confidence = 'NA')
AND detection_type LIKE '%seal%'
AND detection_type NOT LIKE 'dead%'
AND detection_type NOT LIKE 'unknown%'
AND project_schema = 'surv_ice_seals_2025' AND processing_step = 'Species ID (RGB)'
ORDER BY flight, camera_view;

--Check for off-frame RGB data
SELECT flight, camera_view, detection, image_name, detection_type, processing_completed_by
FROM surv_ice_seals_2025.tbl_detections_processed_rgb
INNER JOIN annotations.summ_who_did_what USING (flight, camera_view)
WHERE (bound_bottom < 0 
OR bound_top > 9564
OR bound_right < 0
OR bound_left > 12768)
AND detection_type <> 'false_positive' 
AND detection_type <> 'off_frame'
AND project_schema = 'surv_ice_seals_2025' AND processing_step = 'Species ID (RGB)'
ORDER BY flight, camera_view;

SELECT r.flight, r.camera_view, r.detection, i.detection_type as ir_detection, r.detection_type as rgb_detection, processing_completed_by
FROM surv_ice_seals_2025.tbl_detections_processed_rgb r
INNER JOIN annotations.summ_who_did_what USING (flight, camera_view)
INNER JOIN surv_ice_seals_2025.tbl_detections_processed_ir i USING (processed_detection_id)
WHERE i.detection_type LIKE '%animal%'
AND r.detection_type NOT LIKE '%seal%'
AND r.detection_type NOT LIKE '%bird%'
AND r.detection_type NOT LIKE '%walrus%'
AND r.detection_type NOT LIKE '%animal%'
AND r.detection_type NOT LIKE '%fox%'
AND r.detection_type NOT LIKE '%caribou%'
AND r.detection_type NOT LIKE '%bear%'
AND project_schema = 'surv_ice_seals_2025' AND processing_step = 'Species ID (RGB)'
ORDER BY i.detection_type, r.detection_type;
